-- ----------------------------------------------------------------
-- TRIGGER 1: ACCOMMODATION AMOUNT DERIVATION
-- ----------------------------------------------------------------
-- While inserting/updating an accommodation record, the caller does
-- not need to calculate the amount themselves -- this trigger
-- overwrites whatever value was supplied (even if NULL), computing
-- it from the room's nightly rate and the number of nights stayed.
-- ----------------------------------------------------------------

CREATE OR REPLACE FUNCTION calculate_accommodation_amount()
RETURNS TRIGGER AS $$
DECLARE
    room_rate NUMERIC(12,2);
    num_nights INT;
BEGIN

    SELECT rate_per_night
    INTO room_rate
    FROM hostel_room
    WHERE room_id = NEW.room_id;

    num_nights :=
        NEW.check_out_date::date
        - NEW.check_in_date_time::date;

    IF num_nights <= 0 THEN
        RAISE EXCEPTION
        'Accommodation must be booked for at least one night';
    END IF;

    NEW.amount := room_rate * num_nights;

    RETURN NEW;

END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_calculate_accommodation_amount
BEFORE INSERT OR UPDATE
ON accommodation
FOR EACH ROW
EXECUTE FUNCTION calculate_accommodation_amount();


-- ----------------------------------------------------------------
-- TRIGGER 2: ROOM AVAILABILITY / DATE OVERLAP CHECK
-- ----------------------------------------------------------------
-- Before an accommodation record is inserted or updated, this
-- trigger checks whether the target room has at least one vacancy
-- for the requested check-in/check-out window.
-- ----------------------------------------------------------------

CREATE OR REPLACE FUNCTION check_room_availability()
RETURNS TRIGGER AS
$$
DECLARE
    room_cap INT;
    current_occupancy INT;
BEGIN

    SELECT capacity
    INTO room_cap
    FROM hostel_room
    WHERE room_id = NEW.room_id;

    SELECT COUNT(*)
    INTO current_occupancy
    FROM accommodation a
    WHERE a.room_id = NEW.room_id
      AND a.status = 'ACTIVE'

      AND NEW.check_in_date_time < a.check_out_date
      AND NEW.check_out_date > a.check_in_date_time;

    IF TG_OP = 'UPDATE' THEN
        current_occupancy := current_occupancy - 1;
    END IF;

    IF current_occupancy >= room_cap THEN
        RAISE EXCEPTION
        'Room % is full for selected dates',
        NEW.room_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_check_room_availability
ON accommodation;

CREATE TRIGGER trg_check_room_availability
BEFORE INSERT OR UPDATE
ON accommodation
FOR EACH ROW
EXECUTE FUNCTION check_room_availability();


-- ----------------------------------------------------------------
-- TRIGGER 3: VENUE SCHEDULE CONFLICT CHECK
-- ----------------------------------------------------------------
-- Before an event record is inserted or updated, this trigger
-- checks whether the selected venue is free during the requested
-- date and time window.
-- ----------------------------------------------------------------

CREATE OR REPLACE FUNCTION check_venue_schedule()
RETURNS TRIGGER AS $$
BEGIN

    IF EXISTS (
        SELECT 1
        FROM event e
        WHERE e.venue_id = NEW.venue_id
          AND e.date = NEW.date
          AND e.event_id <> COALESCE(NEW.event_id, -1)

          AND NEW.start_time < e.end_time
          AND NEW.end_time > e.start_time
    )
    THEN
        RAISE EXCEPTION
        'Venue already occupied during selected time slot';
    END IF;

    RETURN NEW;

END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_venue_schedule
BEFORE INSERT OR UPDATE
ON event
FOR EACH ROW
EXECUTE FUNCTION check_venue_schedule();
